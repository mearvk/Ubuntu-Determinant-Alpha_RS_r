# Ubuntu White Edition — GRUB theme
#
# White boot menu with dark-gray entries and an Ubuntu-red selection highlight.
# Colors match ../../../theme/white-edition.css:
#   surface #ffffff, text #303030, muted #666666, edge #dddddd, red #e95420
#
# GRUB themes use named colors / hex strings. Fonts must be supplied as .pf2
# and referenced by their internal name; the installer converts a TTF at build
# time (grub-mkfont) and rewrites the font names if a different face is used.
# The defaults below assume DejaVu Sans, which is present on Ubuntu build hosts.

# ---- Global ----
title-text: ""
desktop-color: "#ffffff"
desktop-image: ""
message-color: "#303030"
message-bg-color: "#ffffff"
terminal-box: "terminal_box_*.png"
terminal-font: "Unifont Regular 16"

}

  width           = 50%

  height          = 52%

  item_color              = "#303030"
  selected_item_color     = "#ffffff"
  item_font               = "DejaVu Sans Regular 16"
  selected_item_font      = "DejaVu Sans Bold 16"
  item_height             = 40
  item_padding            = 6
  item_spacing            = 6
  icon_width              = 0
  icon_height             = 0

  # White row backgrounds; Ubuntu-red highlight on the selected entry.

  item_pixmap_style          = "item_*.png"
  selected_item_pixmap_style = "selected_item_*.png"

  scrollbar               = true
  scrollbar_width         = 6
  scrollbar_thumb         = "scrollbar_thumb_*.png"
}
